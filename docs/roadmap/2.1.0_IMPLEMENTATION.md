## 二、数据结构（shared 类型，精确到字段）

### 2.1 `MaterialRef`（挂在 ComponentNode 上）

```ts
/** 黑盒实例的物料引用（v2.1） */
export interface MaterialRef {
  id: string;          // 物料 type，如 "custom-user-card"
  version: string;     // 锁定版本号，如 "1.2.0"（精确锁定，无 range）
  follow?: 'pin' | 'minor' | 'patch';  // 跟随策略，默认 'pin'（可选）
}
```

`ComponentNode` 增补：

```ts
export interface ComponentNode {
  // ...现有字段...
  materialRef?: MaterialRef;   // 存在 = 黑盒实例；不存在 = 原子节点或展开实例
}
```

### 2.2 契约 `MaterialContract`

```ts
export interface MaterialInput {
  name: string;          // 实例 props 上的 key，如 "title"
  label: string;         // 面板显示名，如 "标题"
  type: 'string' | 'number' | 'boolean' | 'expression' | 'data';
  /** 注入定位：快照内节点 id + 字段路径，如 nodeId: "n1", fieldPath: "props.title" */
  nodeId: string;
  fieldPath: string;     // 点路径，如 "props.title" / "config.columns"
  required: boolean;
  default?: unknown;
}

export interface MaterialOutput {
  name: string;          // 事件出口名，如 "submit"
  label: string;
  nodeId: string;        // 快照内事件源节点 id
  event: string;         // 内部事件名，如 "click" / "submit"
}

export interface MaterialContract {
  inputs: MaterialInput[];
  outputs: MaterialOutput[];
}
```

**为什么用 `nodeId + fieldPath` 而非 `nodes[0].props.title`**：快照是树（可能有 children），数组下标不稳定；黑盒模式不展开、快照内 id 固定，按 id 定位最可靠。

### 2.3 版本与 manifest

```ts
export interface MaterialVersion {
  version: string;                 // "1.0.0"（严格 \d+\.\d+\.\d+）
  schema: ComponentNode[];         // 归一化快照（坐标已归一化、id 为种子）
  contract: MaterialContract;      // 契约（inputs 注入点 / outputs 出口）
  contractSignature: string;       // 契约规范化哈希（major 判定硬标准）
  summary: string[];               // 包含组件清单（展示）
  changelog: string;               // 发布时变更摘要（自动生成 + 可编辑）
  publishedAt: string;             // ISO 时间戳
  releasedBy: string;              // 操作人（resolveOperator）
}

export interface MaterialManifest {   // v2.0 基础上扩展
  type: string;                  // custom- 前缀
  label: string;
  icon: string;
  category: MaterialCategory;
  kind: 'atomic' | 'composite';
  defaultLayout: { w: number; h: number };
  /** 当前版本号（manifest.json 存索引，不含 schema） */
  currentVersion: string;
  /** 版本索引：version → { contractSignature, publishedAt, refCount } */
  versions: Record<string, MaterialVersionMeta>;
  refCount: number;              // 扫描得出的被引用实例数（提示用，删除前实时复核）
  createdAt: string;
}
```

### 2.4 存储文件布局

```
backend/storage/materials/
  custom-user-card/
    manifest.json          # MaterialManifest（元数据 + 版本索引，无 schema）
    draft.json             # MaterialVersion（未发布的编辑态，仅一份；不存在=无草稿）
    versions/
      1.0.0.json           # MaterialVersion（不可变快照，发布后永不改写）
      1.1.0.json
```

## 三、后端实现

### 3.1 `MaterialVersionService`（新文件 `backend/src/services/materialVersionService.ts`）

复用 `storageService.ts` 的成熟模式（写队列 + 原子写），但路径是目录结构：

```ts
export class MaterialVersionService {
  // 复用 StorageService.enqueue 语义：按 materialId 串行写队列
  private writeQueues = new Map<string, Promise<unknown>>();

  // —— 文件读写 ——
  getManifest(id: string): Promise<MaterialManifest | null>;        // materials/<id>/manifest.json
  getVersion(id: string, v: string): Promise<MaterialVersion | null>; // versions/<v>.json
  listVersions(id: string): Promise<MaterialVersionMeta[]>;          // 读 manifest.versions 索引
  getDraft(id: string): Promise<MaterialVersion | null>;             // draft.json

  // —— 写（全部走 enqueue）——
  saveDraft(id: string, version: MaterialVersion): Promise<void>;    // 原子写 draft.json
  discardDraft(id: string): Promise<void>;
  /** 发布：draft → versions/<v>.json + 更新 manifest 索引；draft 保留为"下一版起点"或清除 */
  publish(id: string, draft: MaterialVersion, suggested: string): Promise<{ version: string; changelog: string }>;
  setCurrent(id: string, v: string): Promise<void>;                  // rollback 用：改 manifest.currentVersion
  archiveVersion(id: string, v: string): Promise<void>;              // 移入 versions/archive/ 或删除
  fork(id: string, fromVersion: string, newType: string, label: string): Promise<MaterialManifest>;
  deleteMaterial(id: string): Promise<boolean>;
}
```

要点：
- **不可变保证**：`publish()` 写 `versions/<v>.json` 用原子写（临时文件 + rename）；**不存在更新该文件的入口**——Draft 永远只落 `draft.json`，改物料 = 重新打包进 draft → publish 新版本文件
- **写队列 key = materialId**（复用 `enqueue` 模式），同一物料的 publish / draft / rollback 串行
- **文件名安全**：`toSafeFileId` 复用（拒绝 `/ \ ..`，`custom-xxx` 天然合规）

### 3.2 版本号判定（新文件 `backend/src/utils/versionClassify.ts`）

```ts
/** 契约签名：inputs/outputs 规范化（key 排序）后哈希，复用 frontend 的 shortHash 算法 */
export function contractSignature(contract: MaterialContract): string;

/**
 * 判定建议版本类型。规则（简单、可解释）：
 * 1. 契约签名不同 → major（破坏性：注入点/出口变化）
 * 2. 存在 add/remove 结构操作（节点、字段、配置项增删）→ minor
 * 3. 仅 replace（值级变化：文案、样式值、布局）→ patch
 * 4. 完全无差异 → 提示"内容未变化，禁止发布"
 */
export function classifyVersionChange(
  prev: MaterialVersion,
  next: MaterialVersion
): { bump: 'major' | 'minor' | 'patch'; diff: JsonPatchOp[] };

/** 按 bump 推进版本号：1.2.3 → major=2.0.0 / minor=1.3.0 / patch=1.2.4 */
export function suggestVersion(current: string, bump: 'major' | 'minor' | 'patch'): string;
```

结构 diff 细节：
- 用现有 `createPatch(prev.schema, next.schema)`，但 diff 前**归一化 id**：把两棵树的节点 id 替换为 `$n0/$n1...`（按先序编号），否则重新打包导致的 id 漂移会污染 diff
- `hasStructuralOps = ops.some(op => op.op !== 'replace')` → minor；否则 replace-only → patch
- 契约 diff 单独走 `createPatch(prev.contract, next.contract)` 供 major 确认框展示

### 3.3 引用计数与保护

```ts
/** 扫描 backend/storage/pages/*.json，统计引用某物料 id 的实例数（递归 children） */
export async function scanMaterialRefs(materialId: string): Promise<number>;
```

- 删除物料 / 归档版本：控制器先 `scanMaterialRefs` 实时复核 → `refCount > 0` 时返回 `{ blocked: true, refCount, pages: [...] }`
- manifest.json 的 `refCount` 仅为列表展示，删除/归档以实时扫描为准（防并发竞态）
- 发布 major：控制器返回 `{ refCount }`，前端确认框展示"N 个实例将停留旧版本"

## 四、前端实现

### 4.1 版本解析链（渲染 / 出码共用）

```ts
// frontend/src/utils/materialResolver.ts（新文件）
const cache = new Map<string, MaterialVersion>();   // key = `${id}@${version}`

export async function resolveMaterial(ref: MaterialRef): Promise<MaterialVersion | null> {
  const effectiveVersion = await resolveFollowVersion(ref);   // follow 策略：pin → ref.version；minor/patch → 从列表选最新满足版本
  const key = `${ref.id}@${effectiveVersion}`;
  if (cache.has(key)) return cache.get(key)!;
  const version = await http.get(`/materials/${ref.id}/versions/${effectiveVersion}`).then(r => r.data.data);
  if (version) cache.set(key, version);
  return version ?? null;
}

/** 实例 inputs 注入：深拷贝快照 → 按 contract.inputs 写值 */
export function injectInputs(snapshot: ComponentNode[], instance: ComponentNode, contract: MaterialContract): ComponentNode[];

/** 缺版本/物料被删容错：返回占位渲染信息 */
export function resolveMaterialSafe(ref: MaterialRef): { version: MaterialVersion | null; missing: boolean };
```

- **缓存失效**：物料管理器发布/升级操作后 `cache.clear()`（物料数量有限，全清最简）
- **容错**：`resolveMaterialSafe` 返回 `missing` → 渲染占位节点 + 提示"物料 custom-xxx@1.0.0 缺失，已按当前版本回退或占位"（回退逻辑：missing 时尝试 `resolveCurrent(ref.id)`）

### 4.4 发布流程（物料管理器）


## 五、关键算法伪代码

### 5.1 契约签名

```ts
function contractSignature(contract: MaterialContract): string {
  const canon = {
    inputs: contract.inputs.map(i => ({ name: i.name, type: i.type, nodeId: i.nodeId, fieldPath: i.fieldPath, required: i.required }))
      .sort((a, b) => a.name.localeCompare(b.name)),
    outputs: contract.outputs.map(o => ({ name: o.name, nodeId: o.nodeId, event: o.event }))
      .sort((a, b) => a.name.localeCompare(b.name))
  };
  return shortHash(JSON.stringify(canon));   // FNV-1a 6 位 hex，复用 codeGenerator.shortHash
}
```

### 5.2 版本判定

```ts
function classifyVersionChange(prev, next) {
  if (contractSignature(prev.contract) !== contractSignature(next.contract)) return 'major';
  const prevNorm = normalizeIds(prev.schema);   // id → $n0/$n1...（先序）
  const nextNorm = normalizeIds(next.schema);
  const ops = createPatch(prevNorm, nextNorm);
  if (ops.length === 0) return 'none';          // 禁止发布空版本
  return ops.some(op => op.op !== 'replace') ? 'minor' : 'patch';
}
```

### 5.3 inputs 注入

```ts
function injectInputs(snapshot, instance, contract) {
  const clone = deepClone(snapshot);
  for (const input of contract.inputs) {
    if (instance.props[input.name] === undefined) continue;   // 未配置不注入，用快照默认值
    const node = findNodeById(clone, input.nodeId);
    if (!node) continue;                                       // 容错：注入点失效跳过
    const segs = input.fieldPath.split('.');                   // "props.title" → ["props","title"]
    let cursor: any = node;
    for (let i = 0; i < segs.length - 1; i++) {
      if (cursor[segs[i]] === undefined || typeof cursor[segs[i]] !== 'object') break;
      cursor = cursor[segs[i]];
    }
    if (cursor && typeof cursor === 'object') cursor[segs[segs.length - 1]] = instance.props[input.name];
  }
  return clone;
}
```

### 5.4 outputs 转发（渲染时）

```ts
// 内部节点事件 e 触发 → 查找 contract.outputs 中 nodeId+event 匹配的 output
// → 执行 instance.events[output.name] 的动作链（复用 executeActions）
function forwardOutput(node, internalNodeId, eventName) {
  const out = contract.outputs.find(o => o.nodeId === internalNodeId && o.event === eventName);
  if (!out) return;
  triggerNodeEvent(instance, out.name, payload);
}
```
